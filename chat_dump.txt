    ```python
    from pydantic import BaseModel, Field
    from datetime import datetime
    from typing import Optional
    import uuid
    
    class StockBase(BaseModel):
        symbol: str = Field(..., min_length=1, max_length=10)
        name: str
        exchange: Optional[str] = None
        quantity: int = Field(..., gt=0)
        avg_price: float = Field(..., gt=0)
        currency: str = "USD"
        isin: Optional[str] = None
        sector: Optional[str] = None
        notes: Optional[str] = None
    
    class StockCreate(StockBase):
        pass
    
    class StockUpdate(BaseModel):
        symbol: Optional[str] = None
        name: Optional[str] = None
        exchange: Optional[str] = None
        quantity: Optional[int] = None
        avg_price: Optional[float] = None
        currency: Optional[str] = None
        isin: Optional[str] = None
        sector: Optional[str] = None
        notes: Optional[str] = None
        is_active: Optional[bool] = None
    
    class StockInDBBase(StockBase):
        id: uuid.UUID
        current_price: Optional[float] = None
        date_added: datetime
        last_updated: datetime
        is_active: bool
        
        class Config:
            orm_mode = True
    
    class Stock(StockInDBBase):
        pass
    
    class StockSummary(BaseModel):
        total_investment: float
        current_value: float
        total_gain_loss: float
        total_gain_loss_percent: float
        holdings_count: int
    ```
    
    ### 4. Stock Service Layer (`backend/app/services/stock_service.py`)
    ```python
    import yfinance as yf  # or alpha_vantage, investing.com API
    from typing import List, Optional
    from sqlalchemy.orm import Session
    from app.models.stock import Stock
    from app.schemas.stock import StockCreate, StockUpdate
    
    class StockService:
        @staticmethod
        def get_stock(db: Session, stock_id: str):
            return db.query(Stock).filter(Stock.id == stock_id, Stock.is_active == True).first()
        
        @staticmethod
        def get_stocks(db: Session, skip: int = 0, limit: int = 100):
            return db.query(Stock).filter(Stock.is_active == True).offset(skip).limit(limit).all()
        
        @staticmethod
        def create_stock(db: Session, stock: StockCreate):
            db_stock = Stock(**stock.dict())
            # Fetch current price
            db_stock.current_price = StockService._fetch_current_price(db_stock.symbol, db_stock.currency)
            db.add(db_stock)
            db.commit()
            db.refresh(db_stock)
            return db_stock
        
        @staticmethod
        def update_stock(db: Session, stock_id: str, stock: StockUpdate):
            db_stock = StockService.get_stock(db, stock_id)
            if db_stock:
                update_data = stock.dict(exclude_unset=True)
                for field, value in update_data.items():
                    setattr(db_stock, field, value)
                # Update current price if symbol changed
                if 'symbol' in update_data:
                    db_stock.current_price = StockService._fetch_current_price(db_stock.symbol, db_stock.currency)
                db_stock.last_updated = datetime.datetime.utcnow()
                db.commit()
                db.refresh(db_stock)
            return db_stock
        
        @staticmethod
        def delete_stock(db: Session, stock_id: str):
            db_stock = StockService.get_stock(db, stock_id)
            if db_stock:
                db_stock.is_active = False
                db_stock.last_updated = datetime.datetime.utcnow()
                db.commit()
                db.refresh(db_stock)
            return db_stock
        
        @staticmethod
        def _fetch_current_price(symbol: str, currency: str = "USD") -> Optional[float]:
            try:
                # Add .NS for NSE, .BO for BSE if not present and currency is INR
                yf_symbol = symbol
                if currency == "INR" and not (symbol.endswith('.NS') or symbol.endswith('.BO')):
                    yf_symbol = f"{symbol}.NS"  # Default to NSE
                
                ticker = yf.Ticker(yf_symbol)
                data = ticker.history(period="1d")
                if not data.empty:
                    return float(data['Close'].iloc[-1])
            except Exception as e:
                print(f"Error fetching price for {symbol}: {e}")
            return None
        
        @staticmethod
        def update_all_stock_prices(db: Session):
            """Update current prices for all active stocks"""
            stocks = db.query(Stock).filter(Stock.is_active == True).all()
            for stock in stocks:
                stock.current_price = StockService._fetch_current_price(stock.symbol, stock.currency)
                stock.last_updated = datetime.datetime.utcnow()
            db.commit()
            return stocks
        
        @staticmethod
        def get_portfolio_summary(db: Session):
            stocks = db.query(Stock).filter(Stock.is_active == True).all()
            total_investment = sum(s.avg_price * s.quantity for s in stocks)
            current_value = sum((s.current_price or s.avg_price) * s.quantity for s in stocks)
            total_gain_loss = current_value - total_investment
            total_gain_loss_percent = (total_gain_loss / total_investment * 100) if total_investment > 0 else 0
            
            return StockSummary(
                total_investment=total_investment,
                current_value=current_value,
                total_gain_loss=total_gain_loss,
                total_gain_loss_percent=total_gain_loss_percent,
                holdings_count=len(stocks)
            )
    ```
    
    ### 5. Stock API Endpoints (`backend/app/api/v1/endpoints/stock.py`)
    ```python
    from fastapi import APIRouter, Depends, HTTPException, status
    from sqlalchemy.orm import Session
    from typing import List
    from app.db.database import get_db
    from app.services.stock_service import StockService
    from app.schemas.stock import Stock, StockCreate, StockUpdate, StockSummary
    
    router = APIRouter()
    
    @router.get("/", response_model=List[Stock])
    def read_stocks(skip: int = 0, limit: int = 100, db: Session = Depends(get_db)):
        stocks = StockService.get_stocks(db, skip=skip, limit=limit)
        return stocks
    
    @router.post("/", response_model=Stock)
    def create_stock(stock: StockCreate, db: Session = Depends(get_db)):
        return StockService.create_stock(db=db, stock=stock)
    
    @router.get("/{stock_id}", response_model=Stock)
    def read_stock(stock_id: str, db: Session = Depends(get_db)):
        db_stock = StockService.get_stock(db, stock_id=stock_id)
        if db_stock is None:
            raise HTTPException(status_code=404, detail="Stock not found")
        return db_stock
    
    @router.patch("/{stock_id}", response_model=Stock)
    def update_stock(stock_id: str, stock: StockUpdate, db: Session = Depends(get_db)):
        db_stock = StockService.update_stock(db, stock_id=stock_id, stock=stock)
        if db_stock is None:
            raise HTTPException(status_code=404, detail="Stock not found")
        return db_stock
    
    @router.delete("/{stock_id}")
    def delete_stock(stock_id: str, db: Session = Depends(get_db)):
        db_stock = StockService.delete_stock(db, stock_id=stock_id)
        if db_stock is None:
            raise HTTPException(status_code=404, detail="Stock not found")
        return {"message": "Stock deleted successfully"}
    
    @router.get("/summary/", response_model=StockSummary)
    def get_portfolio_summary(db: Session = Depends(get_db)):
        return StockService.get_portfolio_summary(db)
    
    @router.post("/update-prices/")
    def update_all_prices(db: Session = Depends(get_db)):
        updated_stocks = StockService.update_all_stock_prices(db)
        return {"message": f"Updated prices for {len(updated_stocks)} stocks"}
    ```
